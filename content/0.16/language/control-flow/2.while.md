## Basic While

```zig
const std = @import("std");

test "basic while" {
    var i: u32 = 0;
    while (i < 5) {
        i += 1;
    }
    try std.testing.expectEqual(5, i);
}
```

## Continue Expression

The continue expression (after `:`) runs at the end of every iteration, even after `continue`:

```zig
const std = @import("std");

test "while with continue expression" {
    var i: u32 = 0;
    var sum: u32 = 0;
    while (i < 5) : (i += 1) {
        if (i == 2) continue;
        sum += i;
    }
    // i becomes 0,1,2,3,4 -> sum = 0+1+3+4 = 8
    try std.testing.expectEqual(8, sum);
}
```

## Optional Capture

`while` can repeatedly unwrap an optional until `null`:

```zig
const std = @import("std");

fn next(comptime T: type, arr: []const T, idx: *usize) ?T {
    if (idx.* >= arr.len) return null;
    defer idx.* += 1;
    return arr[idx.*];
}

test "while optional capture" {
    const items = [_]u32{ 1, 2, 3 };
    var idx: usize = 0;
    var sum: u32 = 0;
    while (next(u32, &items, &idx)) |val| {
        sum += val;
    }
    try std.testing.expectEqual(6, sum);
}
```

## Error Union Capture

```zig
const std = @import("std");

fn poll(flag: *bool) !u32 {
    if (flag.*) return 42;
    flag.* = true;
    return error.NotReady;
}

test "while error union capture" {
    var ready = false;
    var total: u32 = 0;
    while (poll(&ready)) |val| {
        total += val;
    } else |_| {
        // error branch runs when loop exits with error
    }
    try std.testing.expectEqual(42, total);
}
```

## While as Expression

Loops can yield a value with `break` and have an `else` branch:

```zig
const std = @import("std");

test "while as expression" {
    var i: u32 = 0;
    const found = while (i < 10) : (i += 1) {
        if (i == 7) break true;
    } else false;

    try std.testing.expect(found);
}
```

## Inline While

Use `inline while` to force loop unrolling at compile time:

```zig
const std = @import("std");

test "inline while" {
    comptime var i: u32 = 0;
    var sum: u32 = 0;
    inline while (i < 4) : (i += 1) {
        sum += i;
    }
    try std.testing.expectEqual(6, sum);
}
```
