## Basic Switch

`switch` is an expression — all branches must coerce to a common type:

```zig
const std = @import("std");

test "basic switch" {
    const x: u32 = 2;
    const result = switch (x) {
        0 => "zero",
        1 => "one",
        2 => "two",
        else => "many",
    };
    try std.testing.expectEqualStrings("two", result);
}
```

## Exhaustiveness

Every possible value must be handled. Use `else` as a catch-all:

```zig
const std = @import("std");

test "switch exhaustiveness" {
    const x: u8 = 100;
    const result = switch (x) {
        0...9 => "small",
        10...99 => "medium",
        else => "large",
    };
    try std.testing.expectEqualStrings("large", result);
}
```

For enums, every variant must be covered — no `else` needed if all cases are handled:

```zig
const Color = enum { red, green, blue };

test "switch enum" {
    const c = Color.green;
    const hex = switch (c) {
        .red => "#FF0000",
        .green => "#00FF00",
        .blue => "#0000FF",
    };
    try std.testing.expectEqualStrings("#00FF00", hex);
}
```

## Ranges

Use `...` for inclusive ranges in switch prongs:

```zig
const std = @import("std");

test "switch ranges" {
    const score: u32 = 85;
    const grade = switch (score) {
        0...49 => "F",
        50...69 => "D",
        70...79 => "C",
        80...89 => "B",
        90...100 => "A",
        else => "invalid",
    };
    try std.testing.expectEqualStrings("B", grade);
}
```

## Multiple Items

Comma-separate multiple values in a single prong:

```zig
const std = @import("std");

test "switch multiple items" {
    const c: u8 = 'a';
    const is_vowel = switch (c) {
        'a', 'e', 'i', 'o', 'u' => true,
        else => false,
    };
    try std.testing.expect(is_vowel);
}
```

## Switch with Union

Switch is commonly used with tagged unions to access variant payloads:

```zig
const std = @import("std");

const Value = union(enum) {
    int: i64,
    float: f64,
    text: []const u8,
};

test "switch tagged union" {
    const v = Value{ .float = 3.14 };
    const desc = switch (v) {
        .int => |i| "integer",
        .float => |f| "float",
        .text => |s| "string",
    };
    try std.testing.expectEqualStrings("float", desc);
}
```

## Switch as Statement

When used as a statement, switch discards the result:

```zig
const std = @import("std");

test "switch statement" {
    const x: u32 = 1;
    switch (x) {
        0 => std.debug.print("zero\n", .{}),
        1 => std.debug.print("one\n", .{}),
        else => std.debug.print("other\n", .{}),
    }
}
```
