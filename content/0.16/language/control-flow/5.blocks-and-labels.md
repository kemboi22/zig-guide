## Blocks are expressions

In Zig, `{ … }` is not only a scope — a **labeled block** can produce a value. You name the block and `break :label value` out of it:

```zig test
const std = @import("std");

test "block expression" {
    const result = blk: {
        const x: u32 = 5;
        const y: u32 = 3;
        break :blk x * y;
    };

    try std.testing.expectEqual(@as(u32, 15), result);
}
```

Why bother? Because you get a place for intermediate `const` bindings without introducing a separate function — and without making those names pollute the outer scope.

```zig test
const std = @import("std");

test "block with nested logic" {
    const result = find: {
        for (0..10) |i| {
            if (i == 7) break :find i;
        }
        break :find @as(usize, 0);
    };

    try std.testing.expectEqual(@as(usize, 7), result);
}
```

::callout{type="tip"}
Label names are arbitrary identifiers. Common style: short names like `blk`, `found`, `outer`, or a word describing the result (`sw`, `init`, `parse`).
::

## Labeled loops

Labels on loops let you `break` or `continue` an **outer** loop from nested code — no boolean flags required:

```zig test
const std = @import("std");

test "labeled break from nested loops" {
    var found: bool = false;

    outer: for (0..5) |i| {
        for (0..5) |j| {
            if (i == 2 and j == 3) {
                found = true;
                break :outer;
            }
        }
    }

    try std.testing.expect(found);
}
```

Works the same with `while`:

```zig test
const std = @import("std");

test "labeled while" {
    var i: u32 = 0;
    var sum: u32 = 0;

    outer: while (i < 3) : (i += 1) {
        var j: u32 = 0;
        while (j < 3) : (j += 1) {
            if (i + j > 3) break :outer;
            sum += 1;
        }
    }

    try std.testing.expectEqual(@as(u32, 6), sum);
}
```

## Labeled continue

`continue :label` skips to the next iteration of the **labeled** loop (running its continue expression, if any):

```zig test
const std = @import("std");

test "labeled continue" {
    var sum: u32 = 0;
    var i: u32 = 0;

    outer: while (i < 3) : (i += 1) {
        var j: u32 = 0;
        while (j < 3) : (j += 1) {
            if (j == 1) continue :outer;
            sum += 1;
        }
    }

    // Each outer iteration only runs j == 0; j == 1 continues outer
    try std.testing.expectEqual(@as(u32, 3), sum);
}
```

## Loops as expressions (recap)

`break` may carry a value. Combined with `else`, loops become searches with defaults:

```zig test
const std = @import("std");

test "loop as expression" {
    const items = [_]u32{ 10, 20, 30, 40, 50 };

    const first_big = for (items) |val| {
        if (val > 25) break val;
    } else 0;

    try std.testing.expectEqual(@as(u32, 30), first_big);
}
```

| Path | Result |
| --- | --- |
| `break value` | Loop expression equals `value` |
| Loop finishes, has `else` | Expression equals the `else` body |
| Loop finishes, no `else`, used as expression | Usually a compile error — provide `else` or do not use as expression |

## Labeled switch

You can label a `switch` and `break :label value` from inside a multi-statement prong:

```zig test
const std = @import("std");

test "labeled switch expression" {
    const x: u32 = 0;

    const result = sw: switch (x) {
        0 => {
            break :sw "zero";
        },
        else => "other",
    };

    try std.testing.expectEqualStrings("zero", result);
}
```

## Inline loops

`inline while` and `inline for` **unroll** at compile time. Each iteration becomes its own copy of the body, which enables comptime-only techniques (type-varying tuples, accumulating comptime strings, calling different functions per step).

### Inline for

```zig test
const std = @import("std");

test "inline for unrolls at comptime" {
    const fields = .{ "x", "y", "z" };
    comptime var result: []const u8 = "";

    inline for (fields) |field| {
        result = result ++ field;
    }

    try std.testing.expectEqualStrings("xyz", result);
}
```

Without `inline`, mutating a `comptime var` across runtime iterations would not work the same way — unrolling makes each step a compile-time event.

### Inline for over tuples

```zig test
const std = @import("std");

test "inline for over tuple" {
    const tuple = .{ 42, true, "hello" };
    comptime var count: usize = 0;

    inline for (tuple) |field| {
        _ = field; // type may differ each iteration
        count += 1;
    }

    try std.testing.expectEqual(@as(usize, 3), count);
}
```

### Inline while

```zig test
const std = @import("std");

test "inline while" {
    comptime var i: u32 = 0;
    var sum: u32 = 0;

    inline while (i < 4) : (i += 1) {
        sum += i;
    }

    try std.testing.expectEqual(@as(u32, 6), sum);
}
```

### Function tables via inline for

Because iterations unroll, each can call a different function from a tuple:

```zig test
const std = @import("std");

fn add(a: i32, b: i32) i32 {
    return a + b;
}
fn sub(a: i32, b: i32) i32 {
    return a - b;
}
fn mul(a: i32, b: i32) i32 {
    return a * b;
}

test "inline for with function tuple" {
    const ops = .{ add, sub, mul };
    const a: i32 = 10;
    const b: i32 = 3;
    var results: [3]i32 = undefined;

    inline for (ops, 0..) |fn_ptr, i| {
        results[i] = fn_ptr(a, b);
    }

    try std.testing.expectEqual(@as(i32, 13), results[0]);
    try std.testing.expectEqual(@as(i32, 7), results[1]);
    try std.testing.expectEqual(@as(i32, 30), results[2]);
}
```

::callout{type="warning"}
Inline unrolling duplicates code. Keep bounds small and fixed. For large or runtime-dynamic iteration, use ordinary loops.
::

## Inline switch

### `inline else`

On a tagged union, `inline else` generates a **specialized body per variant** so the payload type is known exactly in each copy:

```zig test
const std = @import("std");

const Node = union(enum) {
    number: i32,
    text: []const u8,
};

fn describe(node: Node) []const u8 {
    return switch (node) {
        inline else => |payload| @typeName(@TypeOf(payload)),
    };
}

test "inline else switch" {
    const a = Node{ .number = 42 };
    const b = Node{ .text = "hi" };

    try std.testing.expectEqualStrings("i32", describe(a));
    try std.testing.expectEqualStrings("[]const u8", describe(b));
}
```

Without `inline`, a single shared body would have to treat payloads through a common lens — you would not get per-variant types.

### Inline prongs

You can mark specific prongs `inline` when you need specialization for a subset of tags:

```zig test
const std = @import("std");

const Value = union(enum) {
    int: i32,
    float: f64,
    text: []const u8,
};

test "inline prong" {
    const v = Value{ .int = 42 };

    const result = switch (v) {
        inline .int, .float => |val| blk: {
            _ = val;
            break :blk "number";
        },
        .text => |_| "string",
    };

    try std.testing.expectEqualStrings("number", result);
}
```

### When to use `inline else`

- You need the **concrete type** of each union payload.
- Handling is structurally the same but types differ.
- You want the compiler to generate the boilerplate of “one prong per tag.”

## Summary

| Feature | Syntax | Purpose |
| --- | --- | --- |
| Block label | `label: { … }` | Scope + named break target |
| Break with value | `break :label value` | Yield from block or outer loop |
| Continue outer | `continue :label` | Next iteration of labeled loop |
| Inline for | `inline for` | Comptime unroll over sequences/tuples |
| Inline while | `inline while` | Comptime unroll while condition holds |
| Inline else | `inline else => \|p\|` | Per-variant specialized switch bodies |
| Inline prong | `inline .a, .b =>` | Specialize selected tags |

## Putting control flow together

You now have the full core toolkit:

1. **`if`** — bool decisions, optional/error captures  
2. **`while`** — condition loops, continue expressions, draining iterators  
3. **`for`** — collections, ranges, multi-sequence walks  
4. **`switch`** — exhaustive multi-way matches  
5. **Labels & inline** — structured escapes and comptime unrolling  

Together with the type system from the previous section, this is enough to read and write real Zig programs confidently. From here, dig into arrays, pointers, structs, enums, unions, optionals, and error handling as dedicated topics — each builds on these same expression-oriented control-flow ideas.
