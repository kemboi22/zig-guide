## Blocks as Expressions

A block `{ ... }` is an expression that returns a value using `break` with a label:

```zig
const std = @import("std");

test "block expression" {
    const result = blk: {
        const x: u32 = 5;
        const y: u32 = 3;
        break :blk x * y;
    };
    try std.testing.expectEqual(15, result);
}
```

Break from a labeled block can also appear inside nested loops or `if` expressions within the block:

```zig
const std = @import("std");

test "block with nested logic" {
    const result = find: {
        for (0..10) |i| {
            if (i == 7) break :find i;
        }
        break :find @as(u32, 0);
    };
    try std.testing.expectEqual(7, result);
}
```

## Labeled Loops

Label loops to `break` or `continue` an outer loop from nested code:

```zig
const std = @import("std");

test "labeled break from nested loops" {
    var found: bool = false;
    outer: for (0..5) |i| {
        for (0..5) |j| {
            if (i == 2 and j == 3) {
                found = true;
                break :outer;
            }
        }
    }
    try std.testing.expect(found);
}
```

Labels work with any loop type:

```zig
const std = @import("std");

test "labeled while" {
    var i: u32 = 0;
    var sum: u32 = 0;
    outer: while (i < 3) : (i += 1) {
        var j: u32 = 0;
        while (j < 3) : (j += 1) {
            if (i + j > 3) break :outer;
            sum += 1;
        }
    }
    try std.testing.expectEqual(6, sum);
}
```

## Labeled Continue

Continue an outer loop from nested code:

```zig
const std = @import("std");

test "labeled continue" {
    var sum: u32 = 0;
    var i: u32 = 0;
    outer: while (i < 3) : (i += 1) {
        var j: u32 = 0;
        while (j < 3) : (j += 1) {
            if (j == 1) continue :outer;
            sum += 1;
        }
    }
    // each outer iteration: only j=0 runs
    // j=1 triggers continue :outer, skipping the rest of inner loop
    try std.testing.expectEqual(3, sum);
}
```

## Loop as Expression with Break Value

`break` accepts a value, making any loop an expression:

```zig
const std = @import("std");

test "loop as expression" {
    const items = [_]u32{ 10, 20, 30, 40, 50 };
    const first_big = for (items) |val| {
        if (val > 25) break val;
    } else 0;

    try std.testing.expectEqual(30, first_big);
}
```

The `else` branch runs when the loop completes without a `break`.

## Labeled Switch

Switch can be labeled so you can return a value from a prong via `break`:

```zig
const std = @import("std");

test "labeled switch expression" {
    var x: u32 = 0;
    const result = sw: switch (x) {
        0 => {
            break :sw "zero";
        },
        else => "other",
    };
    try std.testing.expectEqualStrings("zero", result);
}
```

## Inline Loops

`inline while` and `inline for` unroll the loop body at compile time, duplicating it for each iteration. This enables compile-time metaprogramming.

### Inline For

```zig
const std = @import("std");

test "inline for unrolls at comptime" {
    const fields = .{ "x", "y", "z" };
    comptime var result = "";
    inline for (fields) |field| {
        result = result ++ field;
    }
    try std.testing.expectEqualStrings("xyz", result);
}
```

Without `inline`, the above would fail because `result` is a `comptime` variable and the loop body would need to be evaluable at compile time. `inline for` makes each iteration a separate copy of the body, allowing comptime state to accumulate.

### Inline For with Tuples

Inline for is especially useful for iterating over tuple fields:

```zig
const std = @import("std");

test "inline for over tuple" {
    const tuple = .{ 42, true, "hello" };
    comptime var count: usize = 0;
    inline for (tuple) |field| {
        count += 1;
    }
    try std.testing.expectEqual(3, count);
}
```

### Inline While

```zig
const std = @import("std");

test "inline while" {
    comptime var i: u32 = 0;
    var sum: u32 = 0;
    inline while (i < 4) : (i += 1) {
        sum += i;
    }
    // Unrolled to: sum += 0; sum += 1; sum += 2; sum += 3;
    try std.testing.expectEqual(6, sum);
}
```

### Compile-Time Function References from Inline Loops

Because inline loops unroll, you can call distinct functions from a tuple per iteration:

```zig
const std = @import("std");

fn add(a: i32, b: i32) i32 { return a + b; }
fn sub(a: i32, b: i32) i32 { return a - b; }
fn mul(a: i32, b: i32) i32 { return a * b; }

test "inline for with function tuple" {
    const ops = .{ add, sub, mul };
    const a: i32 = 10;
    const b: i32 = 3;
    var results: [3]i32 = undefined;
    inline for (ops, 0..) |fn_ptr, i| {
        results[i] = fn_ptr(a, b);
    }
    try std.testing.expectEqual(13, results[0]); // add
    try std.testing.expectEqual(7, results[1]);  // sub
    try std.testing.expectEqual(30, results[2]); // mul
}
```

## Inline Switch

### Inline Else in Switch

`inline else` in a switch on a tagged union generates a separate body for each variant at compile time, with each variant's type known:

```zig
const std = @import("std");

const Node = union(enum) {
    number: i32,
    text: []const u8,
};

fn describe(node: Node) []const u8 {
    return switch (node) {
        inline else => |payload| @typeName(@TypeOf(payload)),
    };
}

test "inline else switch" {
    const a = Node{ .number = 42 };
    const b = Node{ .text = "hi" };
    try std.testing.expectEqualStrings("i32", describe(a));
    try std.testing.expectEqualStrings("[]const u8", describe(b));
}
```

Without `inline else`, `@TypeOf(payload)` would be the common union member type, not the specific variant.

### Inline Prongs

Individual inline prongs allow accessing the union's tag in addition to the payload:

```zig
const std = @import("std");

const Value = union(enum) {
    int: i32,
    float: f64,
    text: []const u8,
};

test "inline prong with tag capture" {
    const v = Value{ .int = 42 };
    const result = switch (v) {
        inline .int, .float => |val| "number",
        .text => |s| "string",
    };
    try std.testing.expectEqualStrings("number", result);
}
```

### When to Use Inline Else

Use `inline else` when:
- You need the **specific type** of each union variant, not the common type
- You want to generate specialized code per variant without writing each prong manually
- You're working with a union where all variants share similar handling logic

## Summary

| Feature | Keyword | Purpose |
| --- | --- | --- |
| Block label | `label: { ... }` | Name a block to break with a value |
| Break to label | `break :label` | Exit a labeled block or outer loop |
| Continue to label | `continue :label` | Skip to next iteration of labeled outer loop |
| Inline for | `inline for` | Unroll for loop at compile time |
| Inline while | `inline while` | Unroll while loop at compile time |
| Inline else | `inline else` | Generate per-variant switch code for tagged unions |
| Inline prong | `inline .a, .b` | Access tag and payload in specific switch prongs |
