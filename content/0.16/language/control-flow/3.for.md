## Basic For

`for` loops iterate over arrays, slices, and other iterable types:

```zig
const std = @import("std");

test "basic for" {
    const items = [_]u32{ 1, 2, 3, 4 };
    var sum: u32 = 0;
    for (items) |val| {
        sum += val;
    }
    try std.testing.expectEqual(10, sum);
}
```

## Index Capture

Capture the index with a second variable:

```zig
const std = @import("std");

test "for with index" {
    const items = [_]u32{ 10, 20, 30 };
    for (items, 0..) |val, idx| {
        try std.testing.expectEqual(val, items[idx]);
    }
}
```

The `0..` is a range expression starting from 0. You can also use explicit ranges:

```zig
for (items, 0..items.len) |val, idx| { ... }
```

## Pointer Capture

Use `*` to get a mutable pointer to each element:

```zig
const std = @import("std");

test "for pointer capture" {
    var items = [_]u32{ 1, 2, 3 };
    for (&items) |*ptr| {
        ptr.* *= 2;
    }
    try std.testing.expectEqual(.{ 2, 4, 6 }, items);
}
```

## For as Expression

Like `while`, `for` can yield a value with `break` and has an `else` branch for when the loop completes without breaking:

```zig
const std = @import("std");

test "for as expression" {
    const items = [_]u32{ 1, 2, 3, 4, 5 };
    const found = for (items) |val| {
        if (val == 3) break true;
    } else false;

    try std.testing.expect(found);
}
```

## Multiple Sequences

Iterate over multiple sequences simultaneously:

```zig
const std = @import("std");

test "for multiple sequences" {
    const a = [_]u32{ 1, 2, 3 };
    const b = [_]u32{ 10, 20, 30 };
    var sum: u32 = 0;
    for (a, b) |va, vb| {
        sum += va + vb;
    }
    try std.testing.expectEqual(66, sum);
}
```

The sequences must have the same length.

## Inline For

Use `inline for` to unroll loops at compile time:

```zig
const std = @import("std");

test "inline for" {
    const fields = .{ "a", "b", "c" };
    comptime var result = "";
    inline for (fields) |field| {
        result = result ++ field;
    }
    try std.testing.expectEqualStrings("abc", result);
}
```
